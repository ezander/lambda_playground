# Symbolic aliases for numeric operations (mixin)
# Can be used with Church numerals, but also with any other numerals implementation that defines
# `zero`, `succ`, `plus`, `times` and `pow` (e.g. Scott numerals)

# First test that already defined symbols (zero, plus, succ) are somewhat sensible
:assert plus (succ zero) (succ zero) ≡ succ (succ zero)
:assert plus (succ zero) (succ (succ zero)) ≡ succ (succ (succ zero))

++ := succ
0 := zero

;             1 := ++  0;  2 := ++  1;  3 := ++  2;  4 := ++  3;
 5 := ++  4;  6 := ++  5;  7 := ++  6;  8 := ++  7;  9 := ++  8;
10 := ++  9; 11 := ++ 10; 12 := ++ 11; 13 := ++ 12; 14 := ++ 13;

20 := ++ 19; 21 := ++ 20; 22 := ++ 21; 23 := ++ 22; 24 := ++ 23;
25 := ++ 24; 26 := ++ 25; 27 := ++ 26; 28 := ++ 27; 29 := ++ 28;
30 := ++ 29; 31 := ++ 30; 32 := ++ 31;

# Rename operators (note, that they either have already been normalized, or they they won't do so here either.
# E.g. the curr definitions of Church minus is defined via the Y combinator and is not normalizing
:set no-normalize-defs
+ := plus
* := times
^^ := pow
- := minus
-- := pred
0? := isZero
even? := isEven
odd? := isOdd
<= := lessEq
< := less
>= := greaterEq
> := greater
== := equal
× := times
≤ := lessEq
≥ := greaterEq
≠ a b := not (equal a b)
:set normalize-defs

:infix + * ^^ -
:infix <= >= == < >
:infix × ≤ ≥ ≠


# Verify number table
:assert 1 ≡ ++ 0
:assert 2 ≡ ++ (++ 0)
:assert 4 ≡ ++ (++ (++ (++ 0)))
:assert 8 ≡ * 2 4
:assert 16 ≡ * 2 8
:assert 32 ≡ * 2 16

# Verify operator aliases
:assert + 3 4 ≡ 7
:assert * 3 10 ≡ 30
:assert ^^ 2 5 ≡ 32
:assert 1 + 3 + 4 ≡ 8
:assert 2 * 3 * 4 ≡ 24

# Verify comparisons
:assert[expectTrue  := {2 <= 3, 0 <= 0, 2 == 2, 3 >= 2, 1 >= 1}] expectTrue ≡ true
:assert[expectFalse := {2 >= 3, 0 <  0, 2 == 1, 3 <= 2, 1 >  1}] expectFalse ≡ false

# Verify Unicode aliases (×, ≤, ≥, ≠)
:assert 3 × 4 ≡ 12
:assert[expectTrue  := {2 ≤ 3, 0 ≤ 0, 3 ≥ 2, 1 ≠ 2}] expectTrue ≡ true
:assert[expectFalse := {3 ≤ 2, 2 ≥ 3, 2 ≠ 2}] expectFalse ≡ false
