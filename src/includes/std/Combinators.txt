# Some standard combinators

I := λx. x
K := λx y. x
S f g x := f x (g x)
B f g x := f (g x)
C f x y := f y x

# Curry's Y combinator (used for recursion) has no terminating normal form
:set no-normalize-defs
Y := λf. (λx. f (x x)) (λx. f (x x))
:set normalize-defs

# The Turing combinator Θ is also not normalizing
:set no-normalize-defs
Θ := (λx. λy. y (x x y)) (λx. λy. y (x x y))
:set normalize-defs

# SKI basis equivalences
:assert S K K ≡ I
:assert S K S ≡ I
:assert S (K S) K ≡ B

# Behavioral identities
:assert I x ≡ x
:assert K x y ≡ x
:assert B f g x ≡ f (g x)
:assert C f x y ≡ f y x
:assert S f g x ≡ f x (g x)

# Derived equivalences
:assert S (K f) g x ≡ f (g x)
:assert S (K f) I x ≡ f x
:assert K I x ≡ I

# Fixed-point property for Y: Y f = f (Y f)
_F := K I
:assert Y _F ≡ _F (Y _F)
:assert Y _F ≡ I
:assert Y (K K) ≡ K

# Fixed-point property for Θ: Θ f = f (Θ f)
:assert Θ _F ≡ _F (Θ _F)
:assert Θ _F ≡ I
:assert Θ (K K) ≡ K
