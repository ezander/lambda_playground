# Symbolic aliases for boolean/logic operations (mixin)
⊤ := true
⊥ := false

¬ := not
∧ := and
∨ := or
→ := implies
↔ := iff
⊕ := xor

:infix ∧ ∨ → ↔ ⊕

# Test using all aliases (de Morgan, biconditional, xor)
:assert[p:={⊥,⊤},q:={⊥,⊤}] ¬ (p ∧ q) ≡ (¬ p) ∨ (¬ q)
:assert[p:={⊥,⊤},q:={⊥,⊤}] p ↔ q ≡ (→ p q) ∧ (→ q p)
:assert[p:={⊥,⊤},q:={⊥,⊤}] ¬ (p ↔ q) ≡ p ⊕ q
